if(PONY_ENGINE_AUTO_SELECT_COMPILER_MODULES)
	message(STATUS "Enabling compiler modification modules. Compiler ID: '${CMAKE_CXX_COMPILER_ID}'")

	if(MSVC)
		if(PONY_ENGINE_CORE)
			message(STATUS "Enabling PONY_ENGINE_CORE_MSVC")
			set(PONY_ENGINE_CORE_MSVC ON)
		endif()
	else()
		message(FATAL_ERROR "Unsupported compiler!")
	endif()
endif()

if(PONY_ENGINE_AUTO_SELECT_PLATFORM_MODULES)
	message(STATUS "Enabling platform modification modules. Platform: '${CMAKE_SYSTEM_NAME}'")

	if(CMAKE_SYSTEM_NAME STREQUAL "Windows")
		if(PONY_ENGINE_APPLICATION)
			message(STATUS "Enabling PONY_ENGINE_APPLICATION_WINAPI")
			set(PONY_ENGINE_APPLICATION_WINAPI ON)
		endif()
		if(PONY_ENGINE_APPLICATION_IMPL)
			message(STATUS "Enabling PONY_ENGINE_APPLICATION_IMPL_WINAPI")
			set(PONY_ENGINE_APPLICATION_IMPL_WINAPI ON)
		endif()

		if(PONY_ENGINE_CORE)
			message(STATUS "Enabling PONY_ENGINE_CORE_WINAPI")
			set(PONY_ENGINE_CORE_WINAPI ON)
		endif()

		if(PONY_ENGINE_FILE_IMPL)
			message(STATUS "Enabling PONY_ENGINE_FILE_IMPL_WINAPI")
			set(PONY_ENGINE_FILE_IMPL_WINAPI ON)
		endif()

		if(PONY_ENGINE_RAW_INPUT_KEYBOARD_IMPL)
			message(STATUS "Enabling PONY_ENGINE_RAW_INPUT_KEYBOARD_IMPL_WINAPI")
			set(PONY_ENGINE_RAW_INPUT_KEYBOARD_IMPL_WINAPI ON)
		endif()
		if(PONY_ENGINE_RAW_INPUT_MOUSE_IMPL)
			message(STATUS "Enabling PONY_ENGINE_RAW_INPUT_MOUSE_IMPL_WINAPI")
			set(PONY_ENGINE_RAW_INPUT_MOUSE_IMPL_WINAPI ON)
		endif()
	else()
		message(FATAL_ERROR "Unsupported platform!")
	endif()
endif()
